cmake_minimum_required(VERSION 3.13)
project(media_kit_linux_plugin_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT FLUTTER_ENGINE_SDK)
  message(FATAL_ERROR "Pass the pinned Linux Flutter engine SDK directory.")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
add_library(flutter SHARED IMPORTED GLOBAL)
set_target_properties(flutter PROPERTIES
  IMPORTED_LOCATION "${FLUTTER_ENGINE_SDK}/libflutter_linux_gtk.so"
  INTERFACE_INCLUDE_DIRECTORIES "${FLUTTER_ENGINE_SDK}")
function(apply_standard_settings target)
  target_compile_options(${target} PRIVATE -Wall -Werror -Wno-unused-parameter)
  target_link_libraries(${target} PRIVATE PkgConfig::GTK)
endfunction()
set(MIMALLOC_USE_STATIC_LIBS OFF CACHE BOOL "No allocator override in link smoke")
add_subdirectory(../../libs/linux/media_kit_libs_linux/linux libraries)
add_subdirectory(../../media_kit_video/linux video)
get_target_property(video_libs media_kit_video_plugin LINK_LIBRARIES)
if("PkgConfig::mpv" IN_LIST video_libs)
  message(FATAL_ERROR "Video plugin must not link distro libmpv.")
endif()
install(TARGETS media_kit_video_plugin DESTINATION lib COMPONENT Runtime)
install(FILES "${MEDIA_KIT_LINUX_MPV}/libmpv.so.2" DESTINATION lib COMPONENT Runtime)
